3. Cryptographic Signature Verification

TypeScript / Node.js Implementation

Always use crypto.timingSafeEqual to defend against side-channel timing attacks:

typescript — verify-webhook.ts 
import crypto from "node:crypto";

export function verifyWebhook(
  incomingSignature: string,
  clientId: string,
  clientSecret: string
): boolean {
  if (!incomingSignature || !clientId || !clientSecret) return false;

  const expectedHash = crypto
    .createHmac("sha256", clientSecret)
    .update(clientId)
    .digest("hex");

  const expectedBuffer = Buffer.from(expectedHash.toLowerCase(), "utf8");
  const incomingBuffer = Buffer.from(incomingSignature.trim().toLowerCase(), "utf8");

  if (expectedBuffer.length !== incomingBuffer.length) return false;
  return crypto.timingSafeEqual(expectedBuffer, incomingBuffer);
}

Python / Flask Implementation

python — app.py 
import hmac
import hashlib
from flask import Flask, request, jsonify, Response

app = Flask(__name__)
CLIENT_ID = "YOUR_CLIENT_ID"
CLIENT_SECRET = "YOUR_CLIENT_SECRET"

@app.route("/api/cowrywise-webhook", methods=["POST"])
def handle_webhook():
    payload = request.get_json()
    signature = payload.get("signature")
    if not signature:
        return Response("Missing signature", status=400)

    computed = hmac.new(
        CLIENT_SECRET.encode("utf-8"),
        CLIENT_ID.encode("utf-8"),
        hashlib.sha256
    ).hexdigest()

    if not hmac.compare_digest(computed.lower(), signature.lower()):
        return Response("Invalid signature", status=401)

    # Process event asynchronously...
    return jsonify({"status": "acknowledged"}), 200

4. Production Engineering Best Practices

  1. Acknowledge Fast (HTTP 200 within 3 seconds): Do not perform heavy database aggregations or external network calls directly inside the HTTP webhook endpoint. Immediately enqueue the event into BullMQ, Celery, AWS SQS, or RabbitMQ and return 200 OK.
  2. Idempotency via Unique Transaction References: Network retries may cause duplicate delivery of webhooks. Store processed transaction references (e.g. event.data.reference) in Redis or a PostgreSQL table with a UNIQUE constraint. If already seen, return 200 OK immediately without re-crediting balances.
  3. Local Testing with Tunneling: Use tools like Ngrok or LocalTunnel to test Cowrywise webhooks on your local workstation:
    bash — tunnel command 
    npx localtunnel --port 3000
# or using ngrok:
ngrok http 3000
